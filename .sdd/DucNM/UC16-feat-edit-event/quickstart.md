## Step 2: Backend Implementation (6-8 hours)

### 2.1 Update Zod Validator

**File**: `backend/src/validators/event.validator.js`

Add `updateEventSchema`:

```javascript
import { z } from 'zod';

export const updateEventSchema = z.object({
  title: z.string()
    .min(10, 'Title must be at least 10 characters')
    .max(500, 'Title must not exceed 500 characters')
    .optional(),
  
  description: z.string()
    .min(50, 'Description must be at least 50 characters')
    .optional(),
  
  location: z.string()
    .min(5, 'Location must be at least 5 characters')
    .max(500, 'Location must not exceed 500 characters')
    .optional(),
  
  start_date: z.string()
    .datetime({ message: 'Invalid datetime format' })
    .refine(
      (val) => new Date(val) >= new Date(),
      { message: 'Start date cannot be in the past' }
    )
    .optional(),
  
  end_date: z.string()
    .datetime({ message: 'Invalid datetime format' })
    .optional(),
  
  application_deadline: z.string()
    .datetime({ message: 'Invalid datetime format' })
    .optional(),
  
  max_capacity: z.number()
    .int('Max capacity must be an integer')
    .positive('Max capacity must be positive')
    .optional(),
  
  category_id: z.number()
    .int('Category ID must be an integer')
    .positive('Category ID must be positive')
    .optional(),
  
  image: z.any().optional()
}).strict()
.refine(
  (data) => {
    if (data.start_date && data.end_date) {
      return new Date(data.end_date) >= new Date(data.start_date);
    }
    return true;
  },
  {
    message: 'End date must be after or equal to start date',
    path: ['end_date']
  }
)
.refine(
  (data) => {
    if (data.application_deadline && data.start_date) {
      return new Date(data.application_deadline) < new Date(data.start_date);
    }
    return true;
  },
  {
    message: 'Application deadline must be before start date',
    path: ['application_deadline']
  }
);
```

### 2.3 Implement EventController.updateEvent()

**File**: `backend/src/controllers/event.controller.js`

```javascript
/**
 * @swagger
 * /api/v1/events/{id}:
 *   patch:
 *     summary: Update event information
 *     tags: [Events]
 *     security:
 *       - cookieAuth: []
 *     parameters:
 *       - in: path
 *         name: id
 *         required: true
 *         schema:
 *           type: integer
 *     requestBody:
 *       content:
 *         multipart/form-data:
 *           schema:
 *             $ref: '#/components/schemas/UpdateEventRequest'
 *     responses:
 *       200:
 *         description: Event updated successfully
 *       400:
 *         description: Validation error
 *       401:
 *         description: Unauthorized
 *       403:
 *         description: Forbidden
 *       404:
 *         description: Event not found
 *       409:
 *         description: Business rule violation
 */
export const updateEvent = async (req, res, next) => {
  try {
    const eventId = parseInt(req.params.id);
    const staffId = req.user.user_id;
    const organizationId = req.user.organization_id;
    
    // Validate request body
    const validatedData = updateEventSchema.parse(req.body);
    
    // Add uploaded file if exists
    if (req.file) {
      validatedData.image = req.file;
    }
    
    // Call service
    const updatedEvent = await EventService.updateEvent(
      eventId,
      staffId,
      organizationId,
      validatedData
    );
    
    return res.json({
      success: true,
      data: updatedEvent,
      message: 'Event updated successfully'
    });
  } catch (error) {
    next(error);
  }
};
```

## Troubleshooting

### Issue 1: Migration Fails

**Symptom**: `prisma migrate dev` fails with foreign key error

**Solution**:

```bash
# Check if users table exists
mysql -u root -p vms_db -e "SHOW TABLES LIKE 'users';"

# If not, run auth migrations first (Member 1)
cd backend
npx prisma migrate deploy
```

### Issue 2: Image Upload Fails

**Symptom**: 500 error when uploading image

**Solution**:

```bash
# Verify Cloudinary credentials
node -e "console.log(process.env.CLOUDINARY_CLOUD_NAME)"

# Test Cloudinary connection
curl -X POST "https://api.cloudinary.com/v1_1/${CLOUDINARY_CLOUD_NAME}/image/upload" \
  -F "file=@test.jpg" \
  -F "upload_preset=vms_events"
```

### Issue 3: Notification Not Sent

**Symptom**: Event updated but no notification created

**Solution**:

```bash
# Check NotificationService exists
grep -r "sendEventChangeNotification" backend/src/services/

# Check notification_types seed data
mysql -u root -p vms_db -e "SELECT * FROM notification_types WHERE name = 'EVENT_CHANGE';"

# If missing, add seed data:
INSERT INTO notification_types (name, description) 
VALUES ('EVENT_CHANGE', 'Event time/location changed');
```

### Issue 4: Audit Log Not Created

**Symptom**: Event updated but no audit log entries

**Solution**:

```bash
# Check if table exists
mysql -u root -p vms_db -e "SHOW TABLES LIKE 'event_audit_log';"

# Check service layer logging logic
grep -A 20 "CRITICAL_FIELDS" backend/src/services/event.service.js
```
